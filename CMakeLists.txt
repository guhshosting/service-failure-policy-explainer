cmake_minimum_required(VERSION 3.24)
project(service-failure-policy-explainer VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows and MSVC are required")
endif()
option(ENABLE_ANALYZE "MSVC static analysis" OFF)
option(ENABLE_ASAN "Address sanitizer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(policy src/model.cpp src/scm.cpp src/decode.cpp src/format.cpp src/cli.cpp src/windows.cpp)
target_include_directories(policy PUBLIC src)
target_compile_definitions(policy PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(policy PUBLIC /W4 /WX /utf-8 /permissive- /EHsc)
target_link_libraries(policy PUBLIC advapi32)
if(ENABLE_ANALYZE)
  target_compile_options(policy PUBLIC /analyze)
endif()
if(ENABLE_ASAN)
  target_compile_options(policy PUBLIC /fsanitize=address /Zi)
  target_link_options(policy PUBLIC /INCREMENTAL:NO)
endif()
add_executable(service-failure-policy-explainer src/main.cpp)
target_link_libraries(service-failure-policy-explainer PRIVATE policy)
enable_testing()
add_executable(policy-tests tests/tests.cpp)
target_link_libraries(policy-tests PRIVATE policy)
add_test(NAME synthetic COMMAND policy-tests)
add_test(NAME help COMMAND service-failure-policy-explainer --help)
add_test(NAME version COMMAND service-failure-policy-explainer --version)
find_program(PWSH_EXECUTABLE pwsh)
if(PWSH_EXECUTABLE)
  add_test(NAME integration-and-json COMMAND "${PWSH_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/validate.ps1" -Executable "$<TARGET_FILE:service-failure-policy-explainer>" -Tests "$<TARGET_FILE:policy-tests>" -Artifacts "${CMAKE_CURRENT_BINARY_DIR}/validation-$<CONFIG>")
  set_tests_properties(integration-and-json PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
install(TARGETS service-failure-policy-explainer RUNTIME DESTINATION bin)
install(FILES README.md LICENSE DESTINATION share/service-failure-policy-explainer)
